FROM (
  SELECT vbitmap.bm_ids(chunk_id, predicate, payload USING PARAMETERS expr =

         OVER (PARTITION BY chunk_id)
  FROM bitmap_enc.index_auto
  WHERE tenant_id = 20 AND snapshot_id = 1 AND domain = 'person'
